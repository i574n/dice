            ?= all ?|
            | f :: rest => f, replay { all = all; rest = rest }

    | Fresh
    | Slide
    | Recycle

:~ ^ roll_hand \* s {draw_face}. (count : i32) (hand : list face) (source : s) : list face * s =
    ? count <= 0
    ?> hand, source
    |:
        := f, source = draw_face source
        roll_hand (count - 1) (f :: hand) source

:~ ^ roll_slid \* t {number} s {draw_face}. (b : bound t) (count : i32) (hand : list face) (source : s) : t * s =
    ?= hand_value b count hand ?|
    | Some value => value, source
    | None =>
        := f, source = draw_face source
        roll_slid b count (f :: hand) source

:~ ^ roll_fresh \* t {number} s {draw_face}. (b : bound t) (count : i32) (source : s) : t * s =
    :~ ^ read (left : i32) (place : t) (value : t) (source : s) : t * s =
        ? left <= 0
        ?> value, source
        |:
            := f, source = draw_face source
            read (left - 1) (place * 6) (value + digit f * place) source
    := value, source = read count 1 0 source
    ? value < bound_value b ?> value + 1, source |: roll_fresh b count source

:! pool t = { value : t; range : t }

:= empty_pool \* t {number}. () : pool t =
    pool { value = 0; range = 1 }

:= settle \* t {number}. (b : bound t) ((pool { value range }) : pool t) : result t (pool t) =
    := max = bound_value b
    := whole = range / max * max
    ? value < whole ?> Ok (value % max + 1) |: Error (pool { value = value - whole; range = range - whole })

:= with_die \* t {number}. ((pool { value range }) : pool t) (f : face) : pool t =
    ? range > (limit.max : t) / 6
    ?> pool { value = digit f; range = 6 }
    |: pool { value = value * 6 + digit f; range = range * 6 }

:~ ^ roll_recycled \* t {number} s {draw_face}. (b : bound t) (p : pool t) (source : s) : t * s =
    ?= settle b p ?|
    | Ok value => value, source
    | Error p =>
        := f, source = draw_face source
        roll_recycled b (with_die p f) source

:= roll \* t {number} s {draw_face}. (policy : reroll) (b : bound t) (source : s) : t * s =
    := count = dice_count b
    ?= policy ?|
    | Recycle => roll_recycled b (empty_pool ()) source
    | Fresh => roll_fresh b count source
    | Slide =>
        := hand, source = roll_hand count [] source
        roll_slid b count hand source

:= roll_value \* t {number} s {draw_face}. (policy : reroll) (max : t) (source : s) : t =

    | Ok b => fst (roll policy b source)
    | Error e => failwith (bound_error_text e)
